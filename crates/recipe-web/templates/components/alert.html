{% import "components/icons.html" as icons %} {#- The severity comes from code (error, warning, info, success), never from user data. -#} {%- macro severity_icon(severity) -%}{% if severity == "error" %}{{ icons::alert_error() }}{% else if severity == "warning" %}{{ icons::alert_warning() }}{% else if severity == "info" %}{{ icons::alert_info() }}{% else if severity == "success" %}{{ icons::alert_success() }}{% endif %}{%- endmacro -%} {#- A static message with the look of the alert and no role, so a page that loads does not announce it. -#} {%- macro notice(severity, message) -%}
{{ severity_icon(severity) }}{{ message }}
{%- endmacro -%} {#- The markup of notice with a block of the caller in place of one text, for example a heading and a list (design D3). It has no role. -#} {%- macro notice_block(severity) -%}
{{ severity_icon(severity) }}
{{ caller() }}
{%- endmacro -%} {%- macro alert(severity, code, message, reference) -%}
{{ severity_icon(severity) }}{{ code }}{{ message }}{{ reference }}
{%- endmacro -%} {#- The alert with field lines between the sentence and the reference. The field names come from code. With link, each line links to the input with that id. No line gives no list. -#} {#- The text of one line of a list: the code, the subject as data, and the sentence. -#} {%- macro line_text(line) -%}{{ line.code }} {% if !line.subject.is_empty() %}{{ line.subject }} {% endif %}{{ line.message }}{%- endmacro -%} {#- The summary of a list of errors (ADR-0041 d). Only the icon, the code, and the count stand in the element with role alert. With link, a line with a target links to its input. -#} {%- macro summary(severity, code, list, link, reference) -%}
{{ severity_icon(severity) }}{{ code }}{{ list.count }}
    {% for line in list.lines %}
  • {% if link && !line.target.is_empty() %}{{ line_text(line) }}{% else %}{{ line_text(line) }}{% endif %}
    • {% endfor %}{% if !list.more.is_empty() %}
  • {{ list.more }}
    • {% endif %}
{{ reference }}

    {% for line in lines %}
  • {% if link && line.has_input %}{{ line.label }}{% else %}{{ line.label }}{% endif %}
    • {% endfor %}
